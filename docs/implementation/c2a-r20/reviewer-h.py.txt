def make_module(name, source, stack=None):
    module = ModuleType('rev19_' + name)
    module.__file__ = _p.__file__
    sys.modules[module.__name__] = module
    exec(compile(source, module.__name__, 'exec'), module.__dict__)
    worker_root = Path(tempfile.mkdtemp(prefix='worker-' + name + '-', dir=TMP))
    package = worker_root / 'dealroom_sign'
    package.mkdir()
    (package / '__init__.py').write_text('')
    (package / 'placement.py').write_text(source)
    (package / 'placement_worker.py').write_bytes((ROOT / 'backend/dealroom_sign/placement_worker.py').read_bytes())
    module.WORKER_CWD = worker_root
    return module

def run_rows(M, rows, both, allowed, worker=False, signers=None):
    pdf = new18.rows_pdf(rows)
    res = {}
    for n, m in M.items():
        mo = Mock()

        r = (m.place(pdf, s, model_client=mo) if worker else old.r14.in_process(m, pdf, s, mo))
        res[n] = (r, bool(new17.wrong_fields(r, allowed)), mo.call_count)
    return res
