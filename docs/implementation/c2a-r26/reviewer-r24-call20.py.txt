import sys, os, subprocess, time
sys.dont_write_bytecode = True
for k in ['OPENROUTER_API_KEY', 'PLACE_MODEL']: os.environ.pop(k, None)
from pathlib import Path
from types import ModuleType
from collections import Counter
from itertools import product
from unittest.mock import Mock, patch
ROOT = Path('/root/worktrees/send-c1b'); sys.path.insert(0, str(ROOT/'backend'))
from dealroom_sign import placement as _p
from dealroom_sign.tests.test_placement_r14 import in_process
from dealroom_sign.tests import placement_r24_review_cases as R24
from dealroom_sign.tests.test_placement import TWO
rc = R24.rc
def show(c): return subprocess.check_output(['git','show',c+':backend/dealroom_sign/placement.py'],cwd=ROOT).decode()
SRC = dict(r21=show('afb9aa6e'), r24=show('9a91a4b9'))
def mk(n):
    m = ModuleType('rev24_'+n); m.__file__ = _p.__file__; sys.modules[m.__name__] = m
    exec(compile(SRC[n], m.__name__, 'exec'), m.__dict__); return m
M = dict((n, mk(n)) for n in SRC)
forbid = Mock(side_effect=AssertionError('http'))
def k(r, allowed):
    if rc.wrong_fields(r, allowed): return 'W'
    return 'p' if r['fields'] else 'n'
calls = Counter()
def run(pages, people, allowed):
    pdf = rc.pages_pdf(pages); out = []
    for n, m in M.items():
        mo = Mock(); r = in_process(m, pdf, people, mo)
        calls[n] += mo.call_count; out.append(k(r, allowed))
    return ''.join(out)
HLA = R24.HLA
ROLES = [('Company','Consultant'),('Company','Contractor'),('Customer','Supplier'),('Customer','Vendor'),
         ('Provider','Client'),('Company','Member'),('Borrower','Lender'),('Company','Investor'),('Discloser','Recipient')]
C = Counter(); EX = Counter()
with patch('httpx.Client.send', forbid):
    t = time.time()
    for (fr, pr), recital, hform, aff, loc, extra, signer in product(
            ROLES, ('defined', 'defined_address', 'none'), ('colon', 'the', 'none'), ('none', 'uncaptioned', 'caption_role'),
            ('after', 'before'), ('placeholders', 'printed'), ('domain', 'explicit', 'both')):
        if aff == 'none' and loc == 'before': continue
        if recital == 'defined':
            rec = [(80, 'This Agreement is entered into between Fleet Test Co. ("%s") and Harbor Light Analytics, Inc. ("%s").' % (fr, pr), False, 40)]
        elif recital == 'defined_address':
            rec = [(80, 'This Agreement is entered into between Fleet Test Co. ("%s") and Harbor Light Analytics, Inc.,' % fr, False, 40),
                   (94, 'a Delaware corporation with offices at 100 Harbor Way, Boston MA ("%s").' % pr, False, 40)]
        else: rec = []
        ph = dict(colon=pr.upper() + ':', the='THE ' + pr.upper(), none=None)[hform]
        fh = dict(colon=fr.upper() + ':', the='THE ' + fr.upper(), none=None)[hform]
        own = R24.company_block(fh, 'FLEET TEST CO.', 40) + R24.company_block(ph, HLA, 320, extra=extra)
        acap = 'GUARANTOR:' if aff == 'caption_role' else None
        a = R24.company_block(acap, 'HARBOR LIGHT GROUP, INC.', 40, 300 if loc == 'after' else 180) if aff != 'none' else []
        if loc == 'after': pages = [rec, own] + ([a] if a else [])
        else: pages = [a + [(y + 400, tx, ru, *p) for y, tx, ru, *p in rec], own]
        pat = dict(name='Pat Smith', email=rc.PE)
        if signer == 'explicit': pat['company'] = 'Harbor Light Analytics, Inc.'
        people = [TWO[1], pat] if signer == 'both' else [pat]
        t2 = run(pages, people, dict([(rc.PE, (2, 320, 300, 500)), (rc.DE, (2, 40, 300, 220))]))
        C[(aff, t2)] += 1
        if 'W' in t2: EX[(t2, pr, recital, hform, aff, signer)] += 1
    print('elapsed', round(time.time()-t), 'http', forbid.call_count, 'calls', sorted(calls.items()))
for x in sorted(C.items()): print(x)
for x in sorted(EX.items(), key=str): print(' W', x)
