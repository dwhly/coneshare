import json
cmds=[]
for line in open('/root/worktrees/dealroom-sign-edd/docs/edd/briefs/counterparty-c2a-gate3-r24.jsonl'):
    try: o=json.loads(line)
    except: continue
    if not isinstance(o,dict) or not isinstance(o.get('message'),dict): continue
    c=o['message'].get('content')
    if isinstance(c,list):
        for b in c:
            if isinstance(b,dict) and b.get('type')=='tool_use':
                cmds.append(b['input'].get('command') or b['input'].get('content') or '')
for IDX in (23, 24):
    body = cmds[IDX].split("<<'EOF'\n",1)[1].rsplit('\nEOF',1)[0]
    old = "SRC = dict(r21=show('afb9aa6e'), r24=show('9a91a4b9'))"
    n = body.count(old); body = body.replace(old, "SRC = dict(r25=show('701f6aee'), r26=show('f450fdae'))")
    print('IDX', IDX, 'subs', n, flush=True)
    exec(compile(body,'fam','exec'), dict(__name__='__main__'))
