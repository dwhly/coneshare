def own_at(h, co, cap, x, y):
    rows = R24.company_block(h, co, x, y)
    if cap: rows.append((y + 72, cap, False, x))
    return rows
def witness(x, y): return [(y, 'Witness:', True, x, x + 180), (y + 12, 'Name:', False, x)]
def notary(x, y): return [(y - 40, 'NOTARY ACKNOWLEDGMENT', False, x), (y - 26, 'Subscribed and sworn before me this ___ day of ______.', False, x),
                          (y, 'Notary Public:', True, x, x + 200), (y + 12, 'My commission expires:', False, x)]
def ack(x, y): return [(y - 26, 'ACKNOWLEDGED AND AGREED:', False, x), (y, 'By:', True, x, x + 180), (y + 12, 'Name:', False, x), (y + 48, 'Date:', True, x, x + 180)]
FLEET = lambda: R24.company_block('DISCLOSER:', 'FLEET TEST CO.', 40)
def layout(L, own, aff):
    # returns pages, pat allowed
    if L == 'same_page3':   # Fleet | own on p2 row 1, affiliate p2 row 2
        return [FLEET() + own(320, 300) + aff(40, 500)], (1, 320, 300, 500)
    if L == 'own_aff_p3':   # Fleet alone on p2; own + affiliate share p3
        return [FLEET(), own(320, 300) + aff(40, 300)], (2, 320, 300, 500)
    if L == 'aff_alone_wit':  # own on p2 with Fleet; affiliate alone on p3 with a witness
        return [FLEET() + own(320, 300), aff(40, 300) + witness(320, 300)], (1, 320, 300, 500)
    if L == 'aff_p3_notary_p3':  # own+aff on p3; notary on p3
        return [FLEET(), own(320, 300) + aff(40, 300) + notary(40, 560)], (2, 320, 300, 500)
    if L == 'aff_p3_ack_p3':
        return [FLEET(), own(320, 300) + aff(40, 300) + ack(320, 560)], (2, 320, 300, 500)
    if L == 'aff_p3_wit_p2':  # own+aff on p3, witness on Fleet's page
        return [FLEET() + witness(320, 300), own(320, 300) + aff(40, 300)], (2, 320, 300, 500)
