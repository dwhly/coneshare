import json, hashlib
cmds=[]
for line in open('/root/worktrees/dealroom-sign-edd/docs/edd/briefs/counterparty-c2a-gate3-r24.jsonl'):
    try: o=json.loads(line)
    except: continue
    if not isinstance(o,dict): continue
    m=o.get('message') or dict()
    if not isinstance(m,dict): continue
    c=m.get('content')
    if isinstance(c,list):
        for b in c:
            if isinstance(b,dict) and b.get('type')=='tool_use':
                cmds.append(b['input'].get('command') or b['input'].get('content') or '')
body = cmds[19].split("<<'EOF'\n",1)[1].rsplit('\nEOF',1)[0]
print('q3b sha', hashlib.sha256(body.encode()).hexdigest()[:12])
old = "SRC = dict(r21=show('afb9aa6e'), r24=show('9a91a4b9'))"
assert body.count(old)==1
body = body.replace(old, "SRC = dict(r21=show('afb9aa6e'), r24=show('9a91a4b9'), r25=show('701f6aee'))")
# also break down by (aff, pr, hform) for r24->r25 transitions
body = body.replace("        C[(aff, t2)] += 1", "        C[(aff, t2)] += 1; D[(aff, pr, hform, recital, t2[1:])] += 1").replace("C = Counter(); EX = Counter()", "C = Counter(); EX = Counter(); D = Counter()")
body += "\nfor x in sorted(D.items(), key=str):\n    if x[0][-1] in ('pn','Wn'): print(' D', x)\n"
exec(compile(body,'q3b','exec'), dict(__name__='__main__'))
